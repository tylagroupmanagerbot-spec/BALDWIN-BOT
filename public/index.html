try{ require('dotenv').config(); }catch{} const express = require('express'); const fs = require('fs'); const path = require('path'); const pino = require('pino'); const { default: makeWASocket, useMultiFileAuthState, fetchLatestBaileysVersion } = require('@whiskeysockets/baileys'); qrData=null; pairingCode=null; console.log('[CONNECTED]'); try{ require('./main.js')(sock); }catch(e){console.log(e)} } if(u.connection==='close'){ isConnected=false; setTimeout(()=>startBot(currentMode),3000); } }); } app.get('/api/status', (req,res)=>{ res.json({connected:isConnected, status:isConnected?'Connected':currentMode, mode:currentMode, qr:qrData, code:pairingCode, number:phoneNum}); }); app.get('/info', (req,res)=>{ res.json({isConnected, qr:qrData, code:pairingCode, mode:currentMode}); }); app.post('/api/mode', async (req,res)=>{ currentMode = req.body.mode==='qr'?'qr':'pairing'; if(currentMode==='qr' && sock){ try{ sock.ev.emit('connection.update',{qr: null}); }catch{} } if(!sock) startBot(currentMode); res.json({ok:true, mode:currentMode}); }); app.post('/setmode', (req,res)=>{ currentMode = req.body.mode==='qr'?'qr':'pairing'; res.json({ok:true}); }); app.post('/api/pair', async (req,res)=>{ try{ let num = (req.body.phoneNumber||req.body.number||'').replace(/\D/g,''); if(!num) return res.json({error:'Enter number'}); phoneNum=num; if(!sock) await startBot('pairing'); if(!sock.authState.creds.registered){ let code = await sock.requestPairingCode(num); pairingCode=code; return res.json({pairingCode:code, code}); }else{ return res.json({message:'Already paired'}); } }catch(e){ res.json({error:e.message}); } }); app.post('/pair', async (req,res)=>{ req.body.phoneNumber=req.body.number; return app._router.handle(req,res,()=>{},'api'); }); startBot('pairing'); app.listen(PORT,'0.0.0.0',()=>console.log(`BALDWIN running on ${PORT}`));
